  <script>
    function toggleTheme() {
      const body = document.body;
      const currentTheme = body.getAttribute('data-theme');
      body.setAttribute('data-theme', currentTheme === 'dark' ? 'light' : 'dark');
    }

    function resetHome() {
      document.getElementById('searchInput').value = '';
      document.getElementById('results').innerHTML = '';
    }

    function performSearch() {
      const query = document.getElementById('searchInput').value.trim() || "أخبار اليوم";
      const resultsContainer = document.getElementById('results');

      const mockData = [
        {
          title: `آخر التطورات والأخبار العاجلة حول: ${query}`,
          snippet: `تغطية الإخبارية الشاملة والمباشرة حول ${query}. تابع أحدث المستجدات والتقارير الموثوقة على مدار الساعة.`,
          fullText: `تغطية تفصيلية كاملة حول موضوع "${query}". تشهد الأحداث الأخيرة تطورات متسارعة مع متابعة دقيقة من المصادر الرسمية والتحليلات المستفيضة.`
        },
        {
          title: `دليل ومواضيع شاملة لـ ${query}`,
          snippet: `اكتشف كافة التفاصيل والمعلومات الهامة المتعلقة بـ ${query} من مصادر متعددة.`,
          fullText: `تقرير شامل يستعرض جميع جوانب ${query} بالتفصيل مع الحقائق والمعطيات المتاحة.`
        }
      ];

      resultsContainer.innerHTML = mockData.map((item, index) => `
        <div class="result-card" onclick="openModal(${index})">
          <div class="result-title">${item.title}</div>
          <div class="result-snippet">${item.snippet}</div>
        </div>
      `).join('');

      window.currentResults = mockData;
    }

    function openModal(index) {
      const item = window.currentResults[index];
      document.getElementById('modalTitle').innerText = item.title;
      document.getElementById('modalBody').innerText = item.fullText;
      document.getElementById('articleModal').classList.add('active');
    }

    function closeModal() {
      document.getElementById('articleModal').classList.remove('active');
    }
  </script>
